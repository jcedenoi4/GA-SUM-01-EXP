\begin{frame}{Tolerancia a fallos: análisis del sistema}
  \begin{block}{Sistema y evidencia por completar}
    \begin{itemize}
      \item Sistema, servicio y componente analizado: \textit{pendiente de confirmar}.
      \item Fallo que soporta y mecanismo de detección: \textit{pendiente de documentar}.
      \item Recuperación observada y límites: \textit{pendiente de documentar}.
    \end{itemize}
  \end{block}
  \begin{alertblock}{Criterio de análisis}
    Distinguir lo que la documentación confirma de lo que se infiere. No atribuir
    redundancia, replicación, failover ni recuperación al sistema sin evidencia específica.
  \end{alertblock}
\end{frame}

\begin{frame}{Incidente real: causas y recuperación}
  \begin{enumerate}
    \item \textbf{Hecho e impacto:} fecha, servicio afectado, duración y usuarios o
          funciones impactados, con fuente y página o sección.
    \item \textbf{Detección:} señal o alerta que reveló el problema y cuánto tardó
          en detectarse, si la fuente lo informa.
    \item \textbf{Causas:} separar el evento desencadenante de las condiciones
          técnicas contribuyentes; validar la causa raíz con el método de
          \texttt{FORM-AC-01}.
    \item \textbf{Respuesta:} mitigación, recuperación y cambio preventivo
          documentados; señalar qué dato no está disponible.
  \end{enumerate}
\end{frame}
